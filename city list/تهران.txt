    function loadAndRenderTehranProducts() {
      const container = document.getElementById('productsContainer');
      const countEl = document.getElementById('billboardCount');
      if (!container || !countEl) return;
      
      // Load billboards from localStorage
      const billboards = JSON.parse(localStorage.getItem('billboardsData') || '[]');
      
      // Filter only Tehran city billboards
      const tehranBillboards = billboards.filter(b => {
        const city = (b.city || '').toLowerCase();
        return city.includes('تهران') || city.includes('tehran');
      });
      
      countEl.textContent = tehranBillboards.length;
      container.innerHTML = '';
      
      if (tehranBillboards.length === 0) {
        container.innerHTML = `
          <div class="no-products" style="grid-column: 1 / -1; text-align: center; padding: 40px; color: #718096;">
            <i class="fa-solid fa-search" style="font-size: 48px; margin-bottom: 20px; opacity: 0.5;"></i>
            <h3>هیچ تابلو تبلیغاتی برای شهر تهران یافت نشد</h3>
            <p>لطفاً از جدول کل تابلوها، تابلوهای مربوط به تهران را اضافه کنید</p>
          </div>
        `;
        return;
      }
      
      // Render product cards for Tehran billboards
      tehranBillboards.forEach(billboard => {
        const productCard = createProductCard(billboard);
        container.appendChild(productCard);
      });
    }

    function createProductCard(billboard) {
      const col = document.createElement('div');
      col.className = 'col-lg-3 col-md-4 col-sm-6 col-xs-12';
      
      const productCard = billboard.productCard || {};
      const image = productCard.imageDataUrl || '../images/products/h1.jpg';
      const title = productCard.title || billboard.position || billboard.billboardCode || 'بیلبورد';
      const price = productCard.priceNumber || 0;
      const priceText = price ? `${price.toLocaleString('fa-IR')} تومان` : 'قیمت نامشخص';
      
      // Determine status
      const isReserved = billboard.isReserved || billboard.labels?.includes('reserved');
      const isAvailable = billboard.isEmpty || billboard.labels?.includes('available');
      const isInactive = billboard.isInactive || billboard.labels?.includes('inactive');
      
      col.innerHTML = `
        <div class="shoping-card" 
          data-id="${billboard.id || ''}" 
          data-code="${billboard.billboardCode || ''}" 
          data-ownership="${billboard.ownership || ''}" 
          data-length="${billboard.length || ''}" 
          data-width="${billboard.width || ''}" 
          data-square="${billboard.squareMeter || ''}"
          data-city="${billboard.city || ''}"
          data-province="${billboard.province || ''}"
          data-address="${billboard.address || ''}"
          data-description="${billboard.description || ''}"
          style="cursor: pointer;">
          <div class="img-sec">
            <img src="${image}" alt="${title}" />
            ${isReserved ? '<span class="hot-offer">رزرو شده</span>' : ''}
            ${isInactive ? '<span class="inactive-offer">غیرفعال</span>' : ''}
            ${isAvailable ? '<span class="available-offer">موجود</span>' : ''}
          </div>
          <div class="title">${title}</div>
          <div style="display: none;" class="titlee">${title}</div>
          <div class="buttons">
            <div class="right"><span class="price">${priceText}</span></div>
            <div class="left">
              <div class="extend-btn" onclick="event.stopPropagation(); addToCart('${billboard.id || billboard.billboardCode}')">
                <a class="b-text" href="#">خرید</a>
                <a class="b-icon" href="#"><i class="fas fa-shopping-cart"></i></a>
              </div>
            </div>
          </div>
        </div>
      `;
      
      // Add click handler for popup
      col.addEventListener('click', function(e) {
        if (e.target.closest('.extend-btn')) return;
        showProductPopup(billboard);
      });
      
      return col;
    }

    function addToCart(billboardId) {
      try {
        const billboards = JSON.parse(localStorage.getItem('billboardsData') || '[]');
        const billboard = billboards.find(b => String(b.id) === String(billboardId) || String(b.billboardCode) === String(billboardId));
        if (!billboard) return;
        
        const productCard = billboard.productCard || {};
        const item = {
          id: String(billboard.id || billboard.billboardCode || productCard.title || Date.now()),
          title: productCard.title || billboard.position || 'بیلبورد',
          priceNumber: productCard.priceNumber || 0,
          image: productCard.imageDataUrl || '../images/products/h1.jpg',
          qty: 1,
          discountPercent: 0
        };
        
        const raw = localStorage.getItem('cartItems');
        let items = [];
        try { 
          items = JSON.parse(raw || '{}').items || []; 
        } catch {}
        
        const existing = items.find(i => String(i.id) === String(item.id));
        if (existing) { 
          existing.qty += 1; 
        } else { 
          items.push(item); 
        }
        
        localStorage.setItem('cartItems', JSON.stringify({ 
          items, 
          totalNumber: items.reduce((s, i) => s + (Number(i.priceNumber) || 0) * (i.qty || 1), 0), 
          ts: Date.now() 
        }));
        
        // Update cart count
        updateCartCount();
        
        // Open cart panel after adding item
        openCartPanel();
        
        if (typeof showNotification === 'function') {
          showNotification('به سبد اضافه شد', 'success');
        } else {
          alert('به سبد اضافه شد');
        }
      } catch (error) {
        console.error('Error adding to cart:', error);
      }
    }

    function updateCartCount() {
      try {
        const cartData = JSON.parse(localStorage.getItem('cartItems') || '{}');
        const totalItems = cartData.items ? cartData.items.reduce((sum, item) => sum + (item.qty || 0), 0) : 0;
        const cartCount = document.querySelector('.cart-count');
        if (cartCount) {
          cartCount.textContent = totalItems;
        }
      } catch (error) {
        console.error('Error updating cart count:', error);
      }
    }

    // Close popup functionality
    document.addEventListener('click', function(e) {
      const popup = document.getElementById('product-popup');
      const closeBtn = document.querySelector('.close-btn');
      
      if (e.target === popup || e.target === closeBtn) {
        popup.style.display = 'none';
      }
    });

    // Initialize cart count on load
    updateCartCount();

    // Initialize cart panel functionality
    function initializeCartPanel() {
      const cartBtn = document.querySelector('.cart');
      const cartPanel = document.getElementById('cartPanel');
      const cartBackdrop = document.getElementById('cartBackdrop');
      const cartClose = document.getElementById('cartClose');
      
      // Open cart panel
      if (cartBtn) {
        cartBtn.addEventListener('click', function(e) {
          e.preventDefault();
          openCartPanel();
        });
      }
      
      // Close cart panel
      if (cartClose) {
        cartClose.addEventListener('click', function() {
          closeCartPanel();
        });
      }
      
      // Close cart panel when clicking backdrop
      if (cartBackdrop) {
        cartBackdrop.addEventListener('click', function() {
          closeCartPanel();
        });
      }
      
      // Close cart panel with Escape key
      document.addEventListener('keydown', function(e) {
        if (e.key === 'Escape') {
          closeCartPanel();
        }
      });
    }

    // Function to go to cart page
    function goToCartPage() {
      try {
        // Get current cart items
        const cartData = JSON.parse(localStorage.getItem('cartItems') || '{}');
        const items = cartData.items || [];
        
        if (items.length === 0) {
          if (typeof showNotification === 'function') {
            showNotification('سبد خرید خالی است! ابتدا محصولی اضافه کنید', 'error');
          } else {
            alert('سبد خرید خالی است! ابتدا محصولی اضافه کنید');
          }
          return;
        }
        
        // Calculate total
        const totalNumber = items.reduce((sum, item) => sum + (Number(item.priceNumber) || 0) * (item.qty || 1), 0);
        
        // Update cart data with current total
        const updatedCartData = {
          items: items,
          totalNumber: totalNumber,
          ts: Date.now()
        };
        
        // Save updated cart data
        localStorage.setItem('cartItems', JSON.stringify(updatedCartData));
        
        // Close cart panel
        closeCartPanel();
        
        // Redirect to cart page
        console.log('Redirecting to cart page...');
        window.location.href = '../cart/cart.html';
        
      } catch (error) {
        console.error('Error going to cart page:', error);
        if (typeof showNotification === 'function') {
          showNotification('خطا در انتقال به صفحه سبد خرید', 'error');
        } else {
          alert('خطا در انتقال به صفحه سبد خرید');
        }
      }
    }

    // Function to open cart panel
    function openCartPanel() {
      const cartPanel = document.getElementById('cartPanel');
      const cartBackdrop = document.getElementById('cartBackdrop');
      
      if (cartPanel && cartBackdrop) {
        cartPanel.classList.add('active');
        cartBackdrop.classList.add('active');
        cartPanel.setAttribute('aria-hidden', 'false');
      }
    }

    // Function to close cart panel
    function closeCartPanel() {
      const cartPanel = document.getElementById('cartPanel');
      const cartBackdrop = document.getElementById('cartBackdrop');
      
      if (cartPanel && cartBackdrop) {
        cartPanel.classList.remove('active');
        cartBackdrop.classList.remove('active');
        cartPanel.setAttribute('aria-hidden', 'true');
      }
    }
