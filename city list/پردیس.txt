  <script>
    document.addEventListener('DOMContentLoaded', function() {
      if (typeof loadCitiesToNavbar === 'function') loadCitiesToNavbar();
      if (typeof applyStoredUserToNavbar === 'function') applyStoredUserToNavbar();
      if (typeof updateNavigationButton === 'function') updateNavigationButton();
      initializeCartPanel();
      loadAndRenderPardisProducts();
    });

    function loadAndRenderPardisProducts(){
      const container = document.getElementById('productsContainer');
      const countEl = document.getElementById('billboardCount');
      if (!container || !countEl) return;
      const billboards = JSON.parse(localStorage.getItem('billboardsData')||'[]');
      const list = billboards.filter(b=>{
        const c = String(b.city||'').toLowerCase();
        return c.includes('پردیس') || c.includes('pardis');
      });
      countEl.textContent = list.length;
      container.innerHTML = '';
      if (!list.length){
        container.innerHTML = '<div class="col-12" style="text-align:center;padding:40px;color:#718096;">هیچ تابلو تبلیغاتی برای شهر پردیس یافت نشد</div>';
        return;
      }
      list.forEach(b=>container.appendChild(createProductCard(b)));
    }

    function createProductCard(billboard){
      const col = document.createElement('div');
      col.className = 'col-lg-3 col-md-4 col-sm-6 col-xs-12';
      const pc = billboard.productCard||{};
      const image = pc.imageDataUrl || '../images/products/h1.jpg';
      const title = pc.title || billboard.position || billboard.billboardCode || 'بیلبورد';
      const price = pc.priceNumber || 0;
      const priceText = price ? (Number(price).toLocaleString('fa-IR')+' تومان') : 'قیمت نامشخص';
      const isReserved = billboard.isReserved || (billboard.labels||[]).includes('reserved');
      const isAvailable = billboard.isEmpty || (billboard.labels||[]).includes('available');
      const isInactive = billboard.isInactive || (billboard.labels||[]).includes('inactive');
      col.innerHTML = `
        <div class="shoping-card" 
          data-id="${billboard.id||''}" 
          data-code="${billboard.billboardCode||''}" 
          data-ownership="${billboard.ownership||''}" 
          data-length="${billboard.length||''}" 
          data-width="${billboard.width||''}" 
          data-square="${billboard.squareMeter||''}" 
          data-city="${billboard.city||''}" 
          data-province="${billboard.province||''}" 
          data-address="${billboard.address||''}" 
          data-description="${billboard.description||''}">
          <div class="img-sec">
            <img src="${image}" alt="${title}" />
            ${isReserved ? '<span class="hot-offer">رزرو شده</span>' : ''}
            ${isInactive ? '<span class="inactive-offer">غیرفعال</span>' : ''}
            ${isAvailable ? '<span class="available-offer">موجود</span>' : ''}
          </div>
          <div class="title">${title}</div>
          <div class="buttons">
            <div class="right"><span class="price">${priceText}</span></div>
            <div class="left">
              <div class="extend-btn" onclick="event.stopPropagation(); addToCart('${billboard.id || billboard.billboardCode}')">
                <a class="b-text" href="#">خرید</a>
                <a class="b-icon" href="#"><i class="fas fa-shopping-cart"></i></a>
              </div>
            </div>
          </div>
        </div>`;
      col.addEventListener('click', function(e){ if (e.target.closest('.extend-btn')) return; showProductPopup(billboard); });
      return col;
    }

    function showProductPopup(b){
      const pc=b.productCard||{};
      const image=pc.imageDataUrl||'../images/products/h1.jpg';
      const title=pc.title||b.position||b.billboardCode||'بیلبورد';
      const price=pc.priceNumber||0;
      const priceText=price?`${price.toLocaleString('fa-IR')} تومان`:'قیمت نامشخص';
      const popup=document.getElementById('product-popup');
      const info=document.getElementById('popup-info');
      if(!popup||!info) return;
      info.innerHTML=`
        <img src="${image}" alt="${title}" style="max-width:100%;border-radius:8px;margin-bottom:15px;">
        <h2 style="color:#2d3748;margin-bottom:10px;">${title}</h2>
        <p style="font-size:18px;font-weight:bold;color:#e53e3e;margin-bottom:15px;">${priceText}</p>`;
      popup.style.display='block';
    }

    function addToCart(id){
      try{
        const billboards=JSON.parse(localStorage.getItem('billboardsData')||'[]');
        const b=billboards.find(x=>String(x.id)===String(id)||String(x.billboardCode)===String(id));
        if(!b) return;
        const pc=b.productCard||{};
        const item={ id: String(b.id||b.billboardCode||pc.title||Date.now()), title: pc.title||b.position||'بیلبورد', priceNumber: pc.priceNumber||0, image: pc.imageDataUrl||'../images/products/h1.jpg', qty:1, discountPercent:0 };
        const raw=localStorage.getItem('cartItems');
        let items=[]; try{ items=JSON.parse(raw||'{}').items||[] }catch{}
        const ex=items.find(i=>String(i.id)===String(item.id));
        if(ex){ ex.qty+=1 } else { items.push(item) }
        localStorage.setItem('cartItems', JSON.stringify({ items, totalNumber: items.reduce((s,i)=>s+(Number(i.priceNumber)||0)*(i.qty||1),0), ts: Date.now() }));
        updateCartCount(); openCartPanel();
      }catch(e){ console.error(e); }
    }

    function updateCartCount(){ try{ const cartData=JSON.parse(localStorage.getItem('cartItems')||'{}'); const totalItems=cartData.items?cartData.items.reduce((s,i)=>s+(i.qty||0),0):0; const cartCount=document.querySelector('.cart-count'); if(cartCount){ cartCount.textContent=totalItems } }catch(e){} }
    function initializeCartPanel(){ const cartBtn=document.querySelector('.cart'); const cartPanel=document.getElementById('cartPanel'); const cartBackdrop=document.getElementById('cartBackdrop'); const cartClose=document.getElementById('cartClose'); if(cartBtn){ cartBtn.addEventListener('click',function(e){ e.preventDefault(); openCartPanel(); }) } if(cartClose){ cartClose.addEventListener('click',function(){ closeCartPanel(); }) } if(cartBackdrop){ cartBackdrop.addEventListener('click',function(){ closeCartPanel(); }) } document.addEventListener('keydown',function(e){ if(e.key==='Escape'){ closeCartPanel(); } }); }
    function openCartPanel(){ const cartPanel=document.getElementById('cartPanel'); const cartBackdrop=document.getElementById('cartBackdrop'); if(cartPanel&&cartBackdrop){ cartPanel.classList.add('active'); cartBackdrop.classList.add('active'); cartPanel.setAttribute('aria-hidden','false'); } }
    function closeCartPanel(){ const cartPanel=document.getElementById('cartPanel'); const cartBackdrop=document.getElementById('cartBackdrop'); if(cartPanel&&cartBackdrop){ cartPanel.classList.remove('active'); cartBackdrop.classList.remove('active'); cartPanel.setAttribute('aria-hidden','true'); } }
