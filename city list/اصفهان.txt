    document.addEventListener('DOMContentLoaded', function() {

    function loadAndRenderIsfahanProducts(){
      const c=document.getElementById('productsContainer'); const count=document.getElementById('billboardCount'); if(!c||!count) return;
      const data=JSON.parse(localStorage.getItem('billboardsData')||'[]');
      const list=data.filter(b=>{const x=(b.city||'').toLowerCase(); return x.includes('اصفهان')||x.includes('isfahan');});
      count.textContent=list.length; c.innerHTML='';
      if(list.length===0){ c.innerHTML=`<div class=\"no-products\" style=\"grid-column:1 / -1;text-align:center;padding:40px;color:#718096;\"><i class=\"fa-solid fa-search\" style=\"font-size:48px;margin-bottom:20px;opacity:.5;\"></i><h3>هیچ تابلو تبلیغاتی برای شهر اصفهان یافت نشد</h3><p>لطفاً از جدول کل تابلوها، تابلوهای مربوط به اصفهان را اضافه کنید</p></div>`; return; }
      list.forEach(b=>c.appendChild(createProductCard(b)));
    }
    function createProductCard(b){ const col=document.createElement('div'); col.className='col-lg-3 col-md-4 col-sm-6 col-xs-12'; const pc=b.productCard||{}; const image=pc.imageDataUrl||'../images/products/h1.jpg'; const title=pc.title||b.position||b.billboardCode||'بیلبورد'; const price=pc.priceNumber||0; const priceText=price?`${price.toLocaleString('fa-IR')} تومان`:'قیمت نامشخص'; const isReserved=b.isReserved||b.labels?.includes('reserved'); const isAvailable=b.isEmpty||b.labels?.includes('available'); const isInactive=b.isInactive||b.labels?.includes('inactive'); col.innerHTML=`<div class=\"shoping-card\" data-id=\"${b.id||''}\" data-code=\"${b.billboardCode||''}\" data-ownership=\"${b.ownership||''}\" data-length=\"${b.length||''}\" data-width=\"${b.width||''}\" data-square=\"${b.squareMeter||''}\" data-city=\"${b.city||''}\" data-province=\"${b.province||''}\" data-address=\"${b.address||''}\" data-description=\"${b.description||''}\" style=\"cursor:pointer;\"><div class=\"img-sec\"><img src=\"${image}\" alt=\"${title}\" />${isReserved?'<span class=\\"hot-offer\\">رزرو شده</span>':''}${isInactive?'<span class=\\"inactive-offer\\">غیرفعال</span>':''}${isAvailable?'<span class=\\"available-offer\\">موجود</span>':''}</div><div class=\"title\">${title}</div><div style=\"display:none;\" class=\"titlee\">${title}</div><div class=\"buttons\"><div class=\"right\"><span class=\"price\">${priceText}</span></div><div class=\"left\"><div class=\"extend-btn\" onclick=\"event.stopPropagation(); addToCart('${b.id||b.billboardCode}')\"><a class=\"b-text\" href=\"#\">خرید</a><a class=\"b-icon\" href=\"#\"><i class=\"fas fa-shopping-cart\"></i></a></div></div></div></div>`; col.addEventListener('click',function(e){ if(e.target.closest('.extend-btn')) return; showProductPopup(b); }); return col; }
    function showProductPopup(b){ const pc=b.productCard||{}; const image=pc.imageDataUrl||'../images/products/h1.jpg'; const title=pc.title||b.position||b.billboardCode||'بیلبورد'; const price=pc.priceNumber||0; const priceText=price?`${price.toLocaleString('fa-IR')} تومان`:'قیمت نامشخص'; const popup=document.getElementById('product-popup'); const info=document.getElementById('popup-info'); if(!popup||!info) return; info.innerHTML=`<img src=\"${image}\" alt=\"${title}\" style=\"max-width:100%;border-radius:8px;margin-bottom:15px;\"><h2 style=\"color:#2d3748;margin-bottom:10px;\">${title}</h2><p style=\"font-size:18px;font-weight:bold;color:#e53e3e;margin-bottom:15px;\">${priceText}</p><div style=\"text-align:right;color:#4a5568;line-height:1.6;\"><div style=\"margin-bottom:8px;\"><strong>کد تابلو:</strong> ${b.billboardCode||'نامشخص'}</div><div style=\"margin-bottom:8px;\"><strong>شهر:</strong> ${b.city||'نامشخص'}</div><div style=\"margin-bottom:8px;\"><strong>استان:</strong> ${b.province||'نامشخص'}</div><div style=\"margin-bottom:8px;\"><strong>آدرس:</strong> ${b.address||'نامشخص'}</div><div style=\"margin-bottom:8px;\"><strong>مالکیت:</strong> ${b.ownership||'نامشخص'}</div><div style=\"margin-bottom:8px;\"><strong>طول:</strong> ${b.length||'نامشخص'} متر</div><div style=\"margin-bottom:8px;\"><strong>عرض:</strong> ${b.width||'نامشخص'} متر</div><div style=\"margin-bottom:8px;\"><strong>مساحت:</strong> ${b.squareMeter||'نامشخص'} متر مربع</div>${b.description?`<div style=\\"margin-bottom:8px;\\"><strong>توضیحات:</strong> ${b.description}</div>`:''}<div style=\"margin-bottom:8px;\"><strong>وضعیت:</strong> ${b.isReserved?'رزرو شده':b.isInactive?'غیرفعال':b.isEmpty?'موجود':'نامشخص'}</div></div>`; popup.style.display='block'; }
    function addToCart(id){ try{ const data=JSON.parse(localStorage.getItem('billboardsData')||'[]'); const b=data.find(x=>String(x.id)===String(id)||String(x.billboardCode)===String(id)); if(!b) return; const pc=b.productCard||{}; const item={ id:String(b.id||b.billboardCode||pc.title||Date.now()), title:pc.title||b.position||'بیلبورد', priceNumber:pc.priceNumber||0, image:pc.imageDataUrl||'../images/products/h1.jpg', qty:1, discountPercent:0 }; const raw=localStorage.getItem('cartItems'); let items=[]; try{ items=JSON.parse(raw||'{}').items||[] }catch{} const ex=items.find(i=>String(i.id)===String(item.id)); if(ex){ ex.qty+=1 } else { items.push(item) } localStorage.setItem('cartItems', JSON.stringify({ items, totalNumber: items.reduce((s,i)=>s+(Number(i.priceNumber)||0)*(i.qty||1),0), ts: Date.now() })); updateCartCount(); openCartPanel(); if(typeof showNotification==='function'){ showNotification('به سبد اضافه شد','success') } else { alert('به سبد اضافه شد') } }catch(e){ console.error('Error adding to cart:',e) } }
    function updateCartCount(){ try{ const cart=JSON.parse(localStorage.getItem('cartItems')||'{}'); const total=cart.items?cart.items.reduce((s,i)=>s+(i.qty||0),0):0; const el=document.querySelector('.cart-count'); if(el){ el.textContent=total } }catch(e){ console.error('Error updating cart count:',e) } }
    document.addEventListener('click',e=>{ const popup=document.getElementById('product-popup'); const closeBtn=document.querySelector('.close-btn'); if(e.target===popup||e.target===closeBtn){ popup.style.display='none' } }); updateCartCount();
    function initializeCartPanel(){ const cartBtn=document.querySelector('.cart'); const cartPanel=document.getElementById('cartPanel'); const cartBackdrop=document.getElementById('cartBackdrop'); const cartClose=document.getElementById('cartClose'); if(cartBtn){ cartBtn.addEventListener('click',e=>{ e.preventDefault(); openCartPanel(); }) } if(cartClose){ cartClose.addEventListener('click',()=>closeCartPanel()) } if(cartBackdrop){ cartBackdrop.addEventListener('click',()=>closeCartPanel()) } document.addEventListener('keydown',e=>{ if(e.key==='Escape') closeCartPanel() }) }
    function goToCartPage(){ try{ const cart=JSON.parse(localStorage.getItem('cartItems')||'{}'); const items=cart.items||[]; if(items.length===0){ if(typeof showNotification==='function'){ showNotification('سبد خرید خالی است! ابتدا محصولی اضافه کنید','error') } else { alert('سبد خرید خالی است! ابتدا محصولی اضافه کنید') } return } const total=items.reduce((s,i)=>s+(Number(i.priceNumber)||0)*(i.qty||1),0); localStorage.setItem('cartItems', JSON.stringify({ items, totalNumber: total, ts: Date.now() })); closeCartPanel(); window.location.href='../cart/cart.html' }catch(e){ console.error('Error going to cart page:',e); if(typeof showNotification==='function'){ showNotification('خطا در انتقال به صفحه سبد خرید','error') } else { alert('خطا در انتقال به صفحه سبد خرید') } } }
    function openCartPanel(){ const p=document.getElementById('cartPanel'); const b=document.getElementById('cartBackdrop'); if(p&&b){ p.classList.add('active'); b.classList.add('active'); p.setAttribute('aria-hidden','false') } }
    function closeCartPanel(){ const p=document.getElementById('cartPanel'); const b=document.getElementById('cartBackdrop'); if(p&&b){ p.classList.remove('active'); b.classList.remove('active'); p.setAttribute('aria-hidden','true') } }
