	<script>
		// Dynamic city detection and page loading
		document.addEventListener('DOMContentLoaded', function() {
			const urlParams = new URLSearchParams(window.location.search);
			const cityName = urlParams.get('city');
			
			if (cityName) {
				// Update page title and header
				document.title = `بیلبوردهای شهر ${cityName}`;
				document.getElementById('cityTitle').textContent = `بیلبوردهای شهر ${cityName}`;
				
				// Set data attribute for city script
				document.body.setAttribute('data-city', cityName);
				
				// Load city page content if available
				const citySlug = cityName.toLowerCase().replace(/\s+/g, '-');
				const cityPages = JSON.parse(localStorage.getItem('cityPages') || '{}');
				
				if (cityPages[citySlug]) {
					// Page exists, continue with normal loading
					console.log('City page found in localStorage');
				} else {
					// Create new city page
					console.log('Creating new city page for:', cityName);
					// This will be handled by the city script
				}
			} else {
				// No city specified, show error or redirect
				document.getElementById('cityTitle').textContent = 'شهر مشخص نشده است';
				document.getElementById('cardsGrid').innerHTML = '<p>لطفاً از طریق منوی شهرها وارد شوید.</p>';
			}
		});
	</script>
